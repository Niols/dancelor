%% Provides several utilities, including \ifstrempty.
\usepackage{etoolbox}

%% Provides \includegraphics.
\usepackage{graphicx}

%% Title page, simple compare to usual TeX title pages, but with an extra notion
%% of “specificity” (eg. Bb instruments, or bass clef). The `\leavevmode` before
%% `\@author` is a bit of an ugly trick to go to horizontal mode, which prevents
%% `\\` from complaining with ”There's no line here to end“ when `\@author` is empty.
\makeatletter
\def\@specificity{}
\def\title#1{\def\@title{#1}\markboth{#1}{}}
\def\specificity#1{\def\@specificity{#1}}
\renewcommand\maketitle{
  \clearpage%
  \thispagestyle{empty}%
  \null\vfill%
  \begin{center}%
    {\Huge \@title}\\[30mm]%
    {\leavevmode\large \@author}\\%
  \end{center}%
  \vfill%
  \begin{center}%
    {\large \@specificity}%
  \end{center}%
  \vfill\null%
  \clearpage%
}
\makeatother

\renewcommand\part[1]{%
  \clearpage%
  \thispagestyle{norightmark}%
  \addparttotoc{#1}%
  \null\vfill%
  \begin{center}
    {\huge #1}%
  \end{center}
  \vfill\vfill\null%
  \clearpage%
}

%% Blank pages will get a message to explain why they are blank. This macro does
%% nothing when the book is simple, and creates the “blank” page in question otherwise.
\newcommand\blankPageIfNonSimple{%
  \ifsimple%
  \else%
    \clearpage%
    \thispagestyle{norightmarknospecificity}%
    \null\vfill%
    \centering%
    \fbox{
      \begin{tabular}{@{}c@{}}
        This page is left blank\\
        for better alignment\\
        of the next multi-pages set.
      \end{tabular}%
    }%
    \vfill\vfill\null%
    \clearpage%
  \fi%
}

\newcommand\renderSetHeader[3]{%
  \vbox{
    {\linespread{0.8}\selectfont\centering\huge #1\par}
    \ifstrempty{#2}{}{\makebox[\linewidth]{#2}}
    \ifstrempty{#3}{}{\makebox[\linewidth]{#3}}
  }%
}

%% Space between elements of a set. This is 14mm by default but can compress
%% to down to 4mm if this allows reducing the total numbre of pages.
\newcommand\renderSetInterElementsSpace{\vspace{14mm minus 10mm}}
\def\minimalSetInterElementsSpace{4mm}

%% A global array of elements in a set; reset at the beginning of each set.
\newcommand\setResetElements{%
  \directlua{setElements = {}}%
}

%% A counter to generate unique box names for set elements.
\newcount\setElementsCounter

%% Storing an element: increase the counter, make a %% box setElement<N>,
%% store the element in it, and add the box register number (\the\setElement<N>)
%% to the array of set elements.
\newcommand\setStoreElement[1]{%
  \global\advance\setElementsCounter by 1
  \expandafter\newbox\csname setElement\the\setElementsCounter\endcsname
  \expandafter\setbox\csname setElement\the\setElementsCounter\endcsname=#1%
  \directlua{table.insert(setElements, \expandafter\the\csname setElement\the\setElementsCounter\endcsname)}%
}

%% Import functions for solving bin packing.
\directlua{dofile("bin-packing.lua")}

\newcommand\setOutputElements{%
  \directlua{
    local interEltSpace = tex.sp("\minimalSetInterElementsSpace")
    eltHeights = {}
    for i, eltBoxNo in pairs(setElements) do
      eltHeights[i] = tex.box[eltBoxNo].height
    end

    %% Compute the number of pages that this set will span. Add blank page if
    %% the current page is odd, unless the set to come spans only one page in
    %% which case it is allowed to stay anywhere.
    %%
    local numberOfPages = minimalNumberOfPages(eltHeights, interEltSpace, tex.dimen.textheight)

    if numberOfPages > 1 and (math.fmod(\thepage, 2) == 1) then
      tex.print("\\blankPageIfNonSimple{}")
    end

    %% Compute the optimal distribution of this set's elements, then render the
    %% elements, forcing new pages when the distribution calls for it.

    local eltsDistrib = balancedPages(eltHeights, interEltSpace, numberOfPages)

    local pageNo = 1
    local firstOnPage = true

    for _, eltBoxNo in ipairs(setElements) do
      if eltsDistrib[pageNo] <= 0 then
        %% we used all the elements allowed on this page, move on to the next
        tex.print("\\clearpage")
        pageNo = pageNo + 1
        firstOnPage = true
      end

      if firstOnPage then
        firstOnPage = false
      else
        tex.print("\\renderSetInterElementsSpace")
      end

      eltsDistrib[pageNo] = eltsDistrib[pageNo] - 1
      tex.print("\\copy" .. eltBoxNo)
    end
  }
}

\newenvironment{set}[3]{%
  \clearpage%
  \thispagestyle{norightmark}%
  \markright{#1}%
  \addsettotoc{#1}{#3}%
  \setResetElements%
  \setStoreElement{\renderSetHeader{#1}{#2}{#3}}%
}{
  \setOutputElements%
  \clearpage%
}

\newcommand\renderTune[4]{%
  \vbox{
    \makebox[\linewidth]{#1\ifstrempty{#2}{}{ \textit{\small(#2)}}\hfill{}#3}\\[2mm]%
    %% Tune are generated on A4 paper -- 21cm wide -- with a left margin of 1cm
    %% for page numbers -- that is 5% of the whole page. Since we want page
    %% numbers to actually appear in the margin on the website, we make the image
    %% bigger but remove those 5% of margin on the left.
    \hspace*{-.05\linewidth}\makebox[1.05\linewidth]{\includegraphics[width=1.05\linewidth]{#4}}%
  }%
}

\newcommand\tune[4]{%
  \addtunetotoc{#1}{#3}%
  \setStoreElement{\renderTune{#1}{#2}{#3}{#4}}%
}
