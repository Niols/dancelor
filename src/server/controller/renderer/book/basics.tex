%% The default documentclasses only support font sizes up to 12pt. We use the
%% package `extsizes` and its documentclass `extbook`.
\ifsimple%
  \documentclass[14pt]{extarticle}
\else%
  \documentclass[14pt, twoside]{extarticle}
\fi%

%% Custom page geometry. It changes depending on whether the document is
%% two-sided.
\usepackage{geometry}
\geometry{
  a4paper,
  includehead,
  top=1mm,
  headheight=6mm,
  headsep=2mm,
  includefoot,
  bottom=3mm,
  footskip=6mm,
}
\ifsimple%
  \geometry{left=12mm, right=12mm} % sum should match with non-simple
\else%
  \geometry{inner=14mm, outer=10mm} % sum should match with simple
\fi%
% \geometry{showframe} %% shows frame around pages for debugging

%% Font
\usepackage{fontspec}
\defaultfontfeatures{Scale=MatchLowercase}
\setromanfont[Numbers=Uppercase]{Source Sans Pro}

%% Headers and footers

\makeatletter

\def\title#1{\gdef\@title{#1}}

\def\@setTitle{}
\def\setTitle#1{\gdef\@setTitle{#1}}

\def\@specificity{}
\def\specificity#1{\gdef\@specificity{#1}}

\usepackage{fancyhdr}

\fancypagestyle{default}{%
  \fancyhf{}
  \def\headrulewidth{0pt}%
  \def\footrulewidth{0pt}%
  \ifShowRunningHeaders
    \ifsimple
      \fancyhead[L]{\@setTitle}
      \fancyhead[R]{\@title}
      \fancyfoot[L]{\@specificity}
      \fancyfoot[R]{\thepage}
    \else
      \fancyhead[LE]{\@setTitle}
      \fancyhead[RO]{\@title}
      \fancyfoot[RE,LO]{\@specificity}
      \fancyfoot[LE,RO]{\thepage}
    \fi
  \fi
}

\fancypagestyle{setTitleAlsoOnOddPage}[default]{%
  \ifShowRunningHeaders
    \fancyhead[LO]{\@setTitle}
  \fi
}

\fancypagestyle{noSetTitle}[default]{%
  \fancyhead[L]{}%
}

\fancypagestyle{noSetTitleOrSpecificity}[default]{%
  \fancyhead[L]{}%
  \ifsimple
    \fancyfoot[L]{}
  \else
    \fancyfoot[RE,LO]{}
  \fi
}

\pagestyle{default}

\makeatother
